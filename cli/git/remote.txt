# change origin of a repo (similar to forking)
git remote remove origin
git remote add origin git@github.com:joel-muller/hslu-enlab.git
git push --all origin

# list remotes
git remote -v

# change remote
git remote set-url origin git@github.com:joel-muller/pdate.git

git remote add origin ssh://git@codeberg.org/mullerjoel/config.git
git remote set-url --add --push origin ssh://git@codeberg.org/mullerjoel/config.git
git remote set-url --add --push origin git@github.com:mullerjoel/config.git
